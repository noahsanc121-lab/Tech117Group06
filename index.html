Introduction

This project solves a problem where individuals may want to play soccer however are unable to due to physical or environmental limitations. For our project we will be creating a simulation of a penalty shot game as seen in Maitse's "Arduino Goalkeeper Project." This project will initially use an Arduino Uno, a KY-023 Joystick module, and a Servo Motor.

Circuit Diagram

The ultrasonic sensor measures distance; the Arduino activates LEDs and the buzzer according to distance thresholds.

Environmental Impact Assessment

  • Materials: The Arduino Uno is RoHS and WEEE compliant, limiting hazardous substances.
  • Energy Efficiency: The circuit draws minimal power (~50 mA at 5V), ideal for USB or solar power.
  • Reusability: All components are modular and reusable, reducing electronic waste.
  • Packaging & Shipping: Ordering all components from a single supplier helps minimize carbon footprint.

Servo goalie; // Servo motor controlling the goalie

const int lcd_maxlen = 17; // 16 + 1 to ensure teh string is null terminated
const char lcd_header[] = "Player    Goalie";
char lcd_score[lcd_maxlen] = {"\0"};
const int ascii_0 = 48;
int player_score = 0;
int goalie_score = 0;

const int variance = 90;
const int pos = variance; // initial goalie position indicating noon/ or directly vertical on a two dimensional plain/y-axis

//pins
const int player_pin = 6;
const int goalie_pin = 7;
const int motor_pin = 8;
LiquidCrystal lcd(12,11,5,4,3,2);
const int joystick_x = A0; /* analog to be able to get vector from motor*/

/* Need to alter what "pos" is above. ideally pos is set to be directly vertical but I don't know what that is right now */
const int max_motor_angle =  pos+variance;
const int min_motor_angle = pos-variance;

/* function headers */
int set_goalie();
int set_lcd();
int set_score();

void setup() {
  pinMode(joystick_x, INPUT);
  pinMode(player_pin, INPUT);
  pinMode(goalie_pin,INPUT);
  goalie.attach(motor_pin);
  lcd.begin(16,2);
  lcd.print(lcd_header);
  set_score();
  memset(lcd_score, ' ',17);
  lcd_score[16]='\0';
  Serial.begin(9600);
}

void loop() {
  static int goalie_last = 0;
  static int player_last = 0;
  int successful_set = set_goalie();
  int player_button = digitalRead(player_pin);
  int goalie_button = digitalRead(goalie_pin);
  if( player_button && player_last == 0){
    player_score++;
    player_last = 1;
  }
  else if (!player_button&& player_last == 1){
    player_last = 0;
  }
  if(goalie_button && goalie_last == 0){
    goalie_score++;
    goalie_last = 1;
  }
  else if (!goalie_button && goalie_last ==1){
    goalie_last = 0;
  }
  int victor = set_lcd();
  
  
  if (victor == 1){
    // do a bunch of lcd printing and such
    lcd.setCursor(0,0);
    lcd.print("Player win!     ");
    delay(5000);
    goalie_score = 0;
    player_score = 0;
    victor = 0;
  }
  if (victor == 2){
    // do a bunch of lcd printing and such
    lcd.setCursor(0,0);
    lcd.print("Goalie win!      ");
    delay(5000);
    goalie_score = 0;
    player_score = 0;
    victor = 0;
  }
}

int set_goalie(){
  // get vector from joystick
  int angle = analogRead(joystick_x);
  // NEED to relativize the angle to the motor position, need tinkercad inputs and outputs
  Serial.print(angle);
  angle = (( (float)angle / 1023) * (variance*2));
  Serial.print("   ");
  Serial.println( angle);
  /* bound the angle by min and max angle */
  angle = angle < min_motor_angle ? min_motor_angle : angle;
  angle = angle > max_motor_angle ? max_motor_angle : angle;
  
  /* set the angle */
  goalie.write(angle);
  
}

int set_lcd(){
  lcd.setCursor(0,0);
  lcd.print(lcd_header);
  lcd.setCursor(0,1);
  lcd.print(lcd_score);
  // before settings score
  if (goalie_score > 9){
    return 2; // 2 represents goalie victory
  }
  else if (player_score > 9){
    return 1; // 1 represents player victory
  }
  set_score();
  
  return 0; // 0 represents no victor yet
}

int set_score(){
  lcd_score[0] = ascii_0 + player_score; // set first position to show player score
  lcd_score[lcd_maxlen-2] = ascii_0 + goalie_score; // set last positoin to show goalie score
  lcd_score[lcd_maxlen-1] = '\0';
}

Conclusion

The system effectively demonstrates distance-based alerting using Arduino. It’s affordable, educational, and sustainable through reusable components.
