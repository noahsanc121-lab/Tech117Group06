Introduction

Our designed our project to solve a problem where individuals may want to play soccer however are unable to due to physical or environmental limitations. For our project we will be creating a simulation of a penalty shot game as seen in Maitse's "Arduino Goalkeeper Project.” The main portion of the project will initially use an Arduino Uno, a KY-023 Joystick module, and a Servo Motor. Player 1 will control the joystick, moving it left and right. The Arduino will then convert these inputs into movement for the servo motor, thus moving the goalie into a position that hopefully stops Player 2’s shot on net. In addition to our main project we designed There are some possible problems that we were worried may arise when constructing our project. The most pressing issue is setting up a reasonable delay for the input of the servo motor. Having the delay too short would allow Player 1 to rapidly move the goalie back and forth so quickly that it makes it nearly impossible for Player 2 to score. On the other hand, if the delay for the servo motor is too long it would give Player 2 an unfair advantage due to delayed reaction time of the goalie. To fix this problem our group has run multiple tests, increasing and decreasing the delay between input and servo motor movement to maximize fairness for both players. Furthermore, another possible issue we were cautious of was having a faulty goalie design where the servo motor wouldn’t be able to support the weight of the goalie or a goalie that wouldn’t withstand the impact of the ball. If the goalie failed in either of these scenarios, it would result in a catastrophic failure because without the goalie, Player 1 is unable to save the shot. In order to resolve this issue, we first opted to use a lighter bristol board, giving the goalie a rigid base needed to stop the ball from entering the net without being so heavy that it would inhibit the motor's movement. In addition, we swapped the original Rubber Ball to a Ping Pong Ball which would stop the goalie from breaking in half when facing the impact of the shot but still allow Player 2 to have an accurate shot.

Arduino Code

The following Arduino code controls the system, lighting LEDs and activating the buzzer based on distance readings from the HC-SR04 sensor.

 
  
// Collision Warning System with Distance Sensor (HC-SR04)
// Author: Noah Sanci
// March 16th 2026
#include <"Servo.h">
#include <"LiquidCrystal.h">
#include <"string.h">

    * \/ ensure lcd_header is long enough, maybe needs one more space

Servo goalie; // Servo motor controlling the goalie

const int lcd_maxlen = 17; // 16 + 1 to ensure teh string is null terminated
const char lcd_header[] = "Player    Goalie";
char lcd_score[lcd_maxlen] = {"\0"};
const int ascii_0 = 48;
int player_score = 0;
int goalie_score = 0;

const int variance = 90;
const int pos = variance; // initial goalie position indicating noon/ or directly vertical on a two dimensional plain/y-axis

//pins
const int player_pin = 6;
const int goalie_pin = 7;
const int motor_pin = 8;
LiquidCrystal lcd(12,11,5,4,3,2);
const int joystick_x = A0; /* analog to be able to get vector from motor*/

/* Need to alter what "pos" is above. ideally pos is set to be directly vertical but I don't know what that is right now */
const int max_motor_angle =  pos+variance;
const int min_motor_angle = pos-variance;

/* function headers */
int set_goalie();
int set_lcd();
int set_score();

void setup() {
  pinMode(joystick_x, INPUT);
  pinMode(player_pin, INPUT);
  pinMode(goalie_pin,INPUT);
  goalie.attach(motor_pin);
  lcd.begin(16,2);
  lcd.print(lcd_header);
  set_score();
  memset(lcd_score, ' ',17);
  lcd_score[16]='\0';
  Serial.begin(9600);
}

void loop() {
  static int goalie_last = 0;
  static int player_last = 0;
  int successful_set = set_goalie();
  int player_button = digitalRead(player_pin);
  int goalie_button = digitalRead(goalie_pin);
  if( player_button && player_last == 0){
    player_score++;
    player_last = 1;
  }
  else if (!player_button&& player_last == 1){
    player_last = 0;
  }
  if(goalie_button && goalie_last == 0){
    goalie_score++;
    goalie_last = 1;
  }
  else if (!goalie_button && goalie_last ==1){
    goalie_last = 0;
  }
  int victor = set_lcd();
  
  
  if (victor == 1){
    // do a bunch of lcd printing and such
    lcd.setCursor(0,0);
    lcd.print("Player win!     ");
    delay(5000);
    goalie_score = 0;
    player_score = 0;
    victor = 0;
  }
  if (victor == 2){
    // do a bunch of lcd printing and such
    lcd.setCursor(0,0);
    lcd.print("Goalie win!      ");
    delay(5000);
    goalie_score = 0;
    player_score = 0;
    victor = 0;
  }
}

int set_goalie(){
  // get vector from joystick
  int angle = analogRead(joystick_x);
  // NEED to relativize the angle to the motor position, need tinkercad inputs and outputs
  Serial.print(angle);
  angle = (( (float)angle / 1023) * (variance*2));
  Serial.print("   ");
  Serial.println( angle);
  /* bound the angle by min and max angle */
  angle = angle < min_motor_angle ? min_motor_angle : angle;
  angle = angle > max_motor_angle ? max_motor_angle : angle;
  
  /* set the angle */
  goalie.write(angle);
  
}

int set_lcd(){
  lcd.setCursor(0,0);
  lcd.print(lcd_header);
  lcd.setCursor(0,1);
  lcd.print(lcd_score);
  // before settings score
  if (goalie_score > 9){
    return 2; // 2 represents goalie victory
  }
  else if (player_score > 9){
    return 1; // 1 represents player victory
  }
  set_score();
  
  return 0; // 0 represents no victor yet
}

int set_score(){
  lcd_score[0] = ascii_0 + player_score; // set first position to show player score
  lcd_score[lcd_maxlen-2] = ascii_0 + goalie_score; // set last positoin to show goalie score
  lcd_score[lcd_maxlen-1] = '\0';
}
